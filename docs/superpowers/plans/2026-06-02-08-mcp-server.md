# ADA MCP Server Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans.

**Goal:** Build the MCP (Model Context Protocol) server that wraps the REST API, letting AI agents (Cursor, Claude Code, Windsurf) call ad platform operations as tools.

**Architecture:** Vercel HTTP-mode MCP server at `mcp.adplatform.is`. Each tool is a thin wrapper around a REST endpoint. Authentication uses MCP-native bearer tokens that map to platform service-account API keys. Tool descriptions are bilingual (Icelandic primary, English secondary).

**Tech Stack:** Anthropic MCP SDK (TypeScript), Hono (HTTP transport), Vercel.

**Depends on:** Plans #1, #2, #4, #5, #6.

**Companion spec section:** 8.3 (MCP tools), positioning notes in §1 of architecture spec.

```ts
const API_BASE = process.env.ADA_API_BASE ?? 'https://api.adplatform.is';

export class ApiClientError extends Error {
  constructor(
    public status: number,
    public code: string,
    message: string,
  ) {
    super(message);
  }
}

export async function apiCall<T>(
  path: string,
  opts: { method?: string; body?: unknown; apiKey: string },
): Promise<T> {
  const res = await fetch(`${API_BASE}${path}`, {
    method: opts.method ?? 'GET',
    headers: {
      Authorization: `Bearer ${opts.apiKey}`,
      'Content-Type': 'application/json',
    },
    body: opts.body ? JSON.stringify(opts.body) : undefined,
  });
  if (!res.ok) {
    let body: { error?: string; message?: string } = {};
    try {
      body = await res.json();
    } catch {
      /* */
    }
    throw new ApiClientError(res.status, body.error ?? 'unknown', body.message ?? res.statusText);
  }
  return (await res.json()) as T;
}
```

```ts
import type { Context } from 'hono';

export function getApiKey(c: Context): string {
  const header = c.req.header('Authorization');
  if (!header || !header.startsWith('Bearer ')) {
    throw new Error('Missing bearer token in MCP request');
  }
  return header.slice('Bearer '.length).trim();
}
```

```ts
import { Hono } from 'hono';
import { StreamableHTTPServerTransport } from '@modelcontextprotocol/sdk/server/streamableHttp.js';
import { createMcpServer } from './server';
import { getApiKey } from './lib/auth';

export const app = new Hono();

app.get('/healthz', (c) => c.json({ ok: true }));

app.post('/mcp', async (c) => {
  let apiKey: string;
  try {
    apiKey = getApiKey(c);
  } catch {
    return c.json({ error: 'missing_auth' }, 401);
  }
  const server = createMcpServer(apiKey);
  const transport = new StreamableHTTPServerTransport({
    sessionIdGenerator: () => crypto.randomUUID(),
  });
  await server.connect(transport);
  const body = await c.req.text();
  // Streamable HTTP responds via transport.handleRequest
  // For Hono Vercel runtime we treat as request/response wrapper
  return new Response(await transport.handleRequest(body), {
    status: 200,
    headers: { 'Content-Type': 'application/json' },
  });
});

export default app;
```

```ts
import { randomBytes, createHash } from 'crypto';
import { db } from '../lib/firebase';

const KEY_COLLECTION = 'api_keys';

export interface ApiKeyRecord {
  id: string;
  ownerEmail: string;
  hash: string; // sha256 of full key
  scope: 'advertiser' | 'publisher' | 'both';
  createdAt: Date;
  lastUsedAt?: Date;
  revoked: boolean;
}

function hash(key: string): string {
  return createHash('sha256').update(key).digest('hex');
}

export async function issueApiKey(
  ownerEmail: string,
  scope: 'advertiser' | 'publisher' | 'both',
): Promise<{ id: string; key: string }> {
  const id = `ak_${randomBytes(8).toString('hex')}`;
  const secret = randomBytes(24).toString('hex');
  const key = `${id}_${secret}`;
  const record: ApiKeyRecord = {
    id,
    ownerEmail,
    hash: hash(key),
    scope,
    createdAt: new Date(),
    revoked: false,
  };
  await db.collection(KEY_COLLECTION).doc(id).set(record);
  return { id, key };
}

export async function verifyApiKey(key: string): Promise<ApiKeyRecord | null> {
  const idMatch = key.match(/^ak_[a-f0-9]{16}/);
  if (!idMatch) return null;
  const id = idMatch[0];
  const snap = await db.collection(KEY_COLLECTION).doc(id).get();
  if (!snap.exists) return null;
  const record = snap.data() as ApiKeyRecord;
  if (record.revoked) return null;
  if (record.hash !== hash(key)) return null;
  return record;
}

export async function revokeApiKey(id: string): Promise<void> {
  await db.collection(KEY_COLLECTION).doc(id).update({ revoked: true });
}
```

```ts
import { verifyApiKey } from '../services/api-keys';

export function requireAuth(): MiddlewareHandler {
  return async (c, next) => {
    const header = c.req.header('Authorization');
    if (!header || !header.startsWith('Bearer ')) {
      return c.json({ error: 'missing_auth' }, 401);
    }
    const token = header.slice('Bearer '.length).trim();

    // Try API key first (cheaper)
    if (token.startsWith('ak_')) {
      const rec = await verifyApiKey(token);
      if (!rec) return c.json({ error: 'invalid_api_key' }, 401);
      c.set('user', { uid: `apikey:${rec.id}`, email: rec.ownerEmail, admin: false });
      await next();
      return;
    }

    // Fall through to Firebase ID token verification
    try {
      const decoded = await auth.verifyIdToken(token);
      if (!decoded.email) return c.json({ error: 'no_email_claim' }, 401);
      c.set('user', { uid: decoded.uid, email: decoded.email, admin: decoded.admin === true });
      await next();
    } catch {
      return c.json({ error: 'invalid_token' }, 401);
    }
  };
}
```

```ts
export const apiKeysRoutes = new Hono();

apiKeysRoutes.post('/', async (c) => {
  try {
    const user = c.get('user');
    const body = (await c.req.json()) as { scope: 'advertiser' | 'publisher' | 'both' };
    const result = await issueApiKey(user.email, body.scope);
    return c.json(
      {
        apiKeyId: result.id,
        apiKey: result.key,
        warning: 'Geymdu þennan lykil. Hann verður ekki birtur aftur.',
      },
      201,
    );
  } catch (e) {
    return handleError(e, c);
  }
});

apiKeysRoutes.delete('/:id', async (c) => {
  try {
    await revokeApiKey(c.req.param('id'));
    return c.json({ ok: true });
  } catch (e) {
    return handleError(e, c);
  }
});
```
